```ts
export function filterSearchDemoItems(
  query: string,
  excluded: readonly SearchAllergen[]
) {
  const term = query.trim().toLocaleLowerCase();

  return SEARCH_DEMO_ITEMS.filter(
    (item) =>
      item.name.toLocaleLowerCase().includes(term) &&
      (!excluded.length ||
        (item.allergens !== null &&
          !item.allergens.some((allergen) =>
            excluded.includes(allergen)
          )))
  );
}
```
